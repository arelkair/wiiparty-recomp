cmake_minimum_required(VERSION 3.20)
project(wii_recomp CXX)
if(WIN32)
    enable_language(RC)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(WP_GAME "wiiparty" CACHE STRING "Game folder under games/ to build")
set(WP_BUILD "2611-035")
set(WP_FP_FLAGS -ffp-contract=off -fno-fast-math)
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE WP_ENGINE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT WP_ENGINE_COMMIT)
    set(WP_ENGINE_COMMIT "unknown")
endif()
set(WP_DISCORD_APP_ID "1574431988449026118")
option(WP_TRACE_CALLS "Record every guest function entry for diagnostics (slower)" OFF)
option(WP_CONSOLE "Build the game as a console program" OFF)
set(WP_CODE_OPT "-O1" CACHE STRING "Optimization flag for the translated game code")

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

enable_testing()

add_subdirectory(third_party)
add_subdirectory(engine)
add_subdirectory(tests)
add_subdirectory(games/${WP_GAME})
